import numpy as np
def f(w):
    return (w - 2)**2
def grad_f(w):
    return 2*(w - 2)
# 超参数
learning_rate = 0.1
num_iterations = 10  #迭代次数
w = 10.0  # 初始化一个参数
for i in range(num_iterations):
    w = w - learning_rate * grad_f(w)
    print(f"第{i+1}次迭代: w = {w:.4f}, f(w) = {f(w):.4f}")

import math
p = 0.7  # 设定出现正面的概率为 70%
expectation = p   # 期望公式：E[X] = p
variance = p * (1 - p)    # 方差公式
print(f"【抛硬币】正面概率: {p}")
print(f"期望 (平均结果): {expectation}")
print(f"方差 (不确定度): {variance:.2f}\n")

# 概念 2：大模型预测下一个词 (Softmax 算法)
raw_scores = {"The": 2.0, "cat": 1.0, "dog": -1.0}
# 步骤 A：用指数函数 (e^x) 处理每个分数。作用：把所有分变成正数，并拉开差距。
exp_values = {}
for word, score in raw_scores.items():
    exp_values[word] = math.exp(score)
# 步骤 B：计算所有指数值的总和
total_exp = sum(exp_values.values())
# 步骤 C：计算最终概率 (个体值 / 总和)
print("【大模型预测下一个词的概率】")
for word, exp_val in exp_values.items():
    probability = exp_val / total_exp
    print(f"词汇 '{word}' 的概率: {probability * 100:.1f}%")
