class Student:
    # 类属性（所有实例共享）
    school_name = "星辰中学"

    def __init__(self, name, score):
        # 实例属性（每个对象独有）
        self.name = name
        self.score = score

    # ----------------------------------------
    # 0. 普通方法 (需要 self)
    # ----------------------------------------
    def show_info(self):
        """用于展示具体某个学生的信息，必须带 self"""
        print(f"[{self.school_name}] 学生: {self.name}, 分数: {self.score}")

    # ----------------------------------------
    # 1. 类方法 @classmethod (需要 cls)
    # ----------------------------------------
    @classmethod
    def change_school_name(cls, new_name):
        """
        对应图片：用于访问和修改类状态
        第一个参数必须是 cls（代表这个类本身）。
        """
        cls.school_name = new_name
        print(f"⚠️ 通知：全校改名为 {cls.school_name}")

    @classmethod
    def create_from_string(cls, info_str):
        """
        类方法的常见用法：作为“备用构造函数”。
        比如从一串文本 "张三-85" 直接创建实例。
        """
        name, score_str = info_str.split("-")
        # cls(name, int(score_str)) 等价于 Student(...)
        return cls(name, int(score_str)) 

    # ----------------------------------------
    # 2. 静态方法 @staticmethod (不需要 self 也不需要 cls)
    # ----------------------------------------
    @staticmethod
    def is_passing(score):
        """
        对应图片：不需要访问类实例或类本身。
        它只处理传进来的参数，像个独立的工具函数。
        """
        return score >= 60

# ==========================================
# 测试与调用
# ==========================================

# 创建两个实例
stu1 = Student("小明", 90)
stu2 = Student("小红", 55)

print("--- 静态方法测试 ---")
# 对应图片：可以通过类或实例调用
print(f"60分及格吗？(通过类调用): {Student.is_passing(60)}") 
print(f"小红的分数及格吗？(通过实例调用): {stu2.is_passing(stu2.score)}")

print("\n--- 类方法测试 ---")
# 对应图片：可以通过类或实例调用，但通常用类调用
Student.change_school_name("宇宙第一中学") 
stu1.show_info()  # 所有学生的学校名字都变了
stu2.show_info()

print("\n--- 类方法作为构造函数测试 ---")
# 使用类方法从字符串直接解析并创建一个新学生
stu3 = Student.create_from_string("李华-99")
stu3.show_info()

class Dog:
    # -------------------------
    # 【类变量】
    # 定义在方法外部。所有狗狗共享的属性。
    # -------------------------
    species = "犬科"      # 所有狗都是犬科
    total_dogs = 0        # 记录一共实例化了多少条狗

    def __init__(self, name, color):
        # -------------------------
        # 【实例变量】
        # 带有 self. 前缀。每条狗独有的属性。
        # -------------------------
        self.name = name    # 狗的名字
        self.color = color  # 狗的毛色
        
        # 每次创建一只新狗，类变量 total_dogs 就会 +1
        Dog.total_dogs += 1 

# ==========================================
# 测试 1：基本区别
# ==========================================
dog1 = Dog("大黄", "黄色")
dog2 = Dog("小白", "白色")

print("--- 1. 访问变量 ---")
# 实例变量：各自独立
print(f"dog1 的名字: {dog1.name}, 颜色: {dog1.color}") 
print(f"dog2 的名字: {dog2.name}, 颜色: {dog2.color}")

# 类变量：大家共享
print(f"dog1 的物种: {dog1.species}")
print(f"dog2 的物种: {dog2.species}")
print(f"当前世界上总共有 {Dog.total_dogs} 只狗") 


# ==========================================
# 测试 2：修改类变量（正确姿势）
# ==========================================
print("\n--- 2. 修改类变量 ---")
# 正确做法：通过【类名】来修改
Dog.species = "外星犬科" 

# 修改后，所有狗狗的物种全变了！这就叫“牵一发而动全身”
print(f"dog1 现在的物种: {dog1.species}") # 输出: 外星犬科
print(f"dog2 现在的物种: {dog2.species}") # 输出: 外星犬科


# ==========================================
# 测试 3：高危陷阱（通过实例修改类变量）
# ==========================================
print("\n--- 3. 经典陷阱 ---")
# ⚠️ 错误做法：试图通过【实例名】去修改类变量
dog1.species = "机器狗"

# 看看发生了什么神奇的事情：
print(f"dog1 的物种: {dog1.species}") # 输出: 机器狗 (看起改成功了？)
print(f"dog2 的物种: {dog2.species}") # 输出: 外星犬科 (dog2 没受影响！)
print(f"类的物种: {Dog.species}")     # 输出: 外星犬科 (类变量根本没变！)

# 【原理解析】：
# 当你写 `dog1.species = "机器狗"` 时，Python 并没有去改墙上的“公告板”（类变量）。
# 相反，Python 会在 dog1 这条狗的身上，新建一个也叫 "species" 的【实例变量】。
# 此时 dog1 身上有一个实例变量叫 species，类里面也有一个类变量叫 species。
# 当你打印 dog1.species 时，实例变量优先级更高，遮蔽了类变量。

16、

python是按照引用调用还是按照值调用？

可变对象(列表，字典，集合)：在函数内部对可变对象的修改会影响到原始对象。、

不可变对象（整数，字符串，元组）：在函数内部对不可变对象的任何操作不会影响到原始对象，因为这些操作实际上会创建一个新的对象。

当一个对象作为参数传递给函数时，实际上是传递了这个对象的引用。（即对象的内存地址）

如果传递的是一个可变对象，在函数内部对这个对象的修改会影响到原始对象，因为函数内部和外部的变量都指向同一个对象。

如果传递的是一个不可变对象，在函数内部对这个对象的修改不会影响到原始对象，因为任何修改都会创建一个新的对象，而不会改变原有对象。

20、可迭代对象和迭代器

可迭代对象是一个可以返回迭代器的对象，任何实现了iter（）或者getitem()方法的对象都是可迭代的。

只要能放进for in 循环中的给东西，都是可迭代对象。iter（对象）被调用时会有一个迭代器，类似于书签。迭代器是有状态的，他会记住当前遍历到的位置。每次调用next(迭代器)，他就会返回当前位置的值，并把指针移到下一个位置，只能往前走，不能往后退。

for循环的真面目：

1
2
3

my_list = [1, 2, 3, 4]
for num in my_list:
    print(num)

遇到for ，py首先偷偷调用iter（my_list）,获取一个迭代器。然后py启动一个死循环，不断地对这个迭代器调用next()，再把next（）返回的值交给变量num，然后执行print(num),，直到捕获stopiteration报错，数据获取完毕，退出循环。

import asyncio
import time

# --- 模拟耗时的网络请求 ---

# 1. 传统的同步写法（收银员死等）
def sync_request(name):
    print(f"开始处理: {name}")
    time.sleep(2)  # 模拟网络请求耗时 2 秒，此时程序完全卡死
    print(f"完成处理: {name}")

# 2. 异步写法（收银员接单后立刻干别的）
async def async_request(name):
    print(f"开始处理: {name}")
    await asyncio.sleep(2) # 模拟耗时。await 让出控制权，不卡死程序！
    print(f"完成处理: {name}")

# --- 测试运行 ---

async def main():
    start_time = time.time()
    
    # 假设有 3 个任务
    print("--- 异步并发执行 ---")
    # asyncio.gather 相当于让收银员同时处理 3 个单子
    await asyncio.gather(
        async_request("任务A"),
        async_request("任务B"),
        async_request("任务C")
    )
    print(f"异步总耗时: {time.time() - start_time:.2f} 秒\n")

# 运行异步的事件循环
asyncio.run(main())

事件循环：asyncio的核心，负责调度和执行异步任务

协程：使用async def 定义，异步执行的函数，可以在执行过程中挂起和恢复

任务：协程的封装，表示协程的执行，可以由事件循环调度。

异步i/o：通过await关键字等待异步操作，不会阻塞事件循环。

通过使用asyncio，可以编写高效的异步代码，在处理大量i/o操作时显著提高性能和响应速度。

24、元类（metaclass）

如果对象是汽车，类是汽车图纸，元类就是制作图纸的机器。

元类用于创建类的类，定义了类的创建和行为方式。

默认元类是type，通过重载袁磊的new和init方法，可以控制类的创建过程和行为。

如果我们不想用默认的 type 来造类，我们就可以自己写一个元类（继承自 type），并在里面重写这两个魔法方法：

  • new（负责“生”）： 在类被创建出来的那一瞬间调用。它负责在内存里开辟空间，真正把“类”这个实体给捏出来。如果你想拦截类的创建，或者修改类的名字、属性，都在这里做。
  • init（负责“养”）： 在类被 __new__ 捏出来之后调用。类已经存在了，你可以在这里给它额外绑定一些属性或做些初始化操作。

# 1. 定义元类：插件注册机
class PluginMeta(type):
    # 准备一个字典，作为所有插件的“花名册”
    registry = {}

    def __new__(cls, name, bases, attrs):
        # cls: 元类本身 (PluginMeta)
        # name: 正在创建的类的名字 (比如 'AudioPlugin')
        # bases: 正在创建的类继承了谁
        # attrs: 正在创建的类里面写了哪些属性和方法
        
        # 先让老祖宗 type 把这个类给正常造出来
        new_class = super().__new__(cls, name, bases, attrs)
        
        # 如果这个类不是基类本身，就把它登记到花名册里
        if name != "BasePlugin":
            cls.registry[name] = new_class
            print(f"✅ 自动拦截并注册了新插件: {name}")
            
        return new_class

# 2. 定义一个基类，并指定它的元类是我们的“注册机”
class BasePlugin(metaclass=PluginMeta):
    pass

print("--- 开始定义具体的插件 ---")

# 3. 程序员只需要正常写代码，继承 BasePlugin 即可
class AudioPlugin(BasePlugin):
    def play(self):
        print("播放声音")

class VideoPlugin(BasePlugin):
    def play(self):
        print("播放视频")

class SubtitlePlugin(BasePlugin):
    def show(self):
        print("显示字幕")

print("\n--- 查看最终的花名册 ---")
# 此时，我们没有写任何手动注册的代码，但 PluginMeta 已经在后台帮我们搞定了
print(PluginMeta.registry)

import time

# 定义一个用于计时的装饰器
def timer_decorator(func):
    def wrapper():
        start_time = time.time()  # 记录开始时间
        
        func()                    # 执行原本的函数
        
        end_time = time.time()    # 记录结束时间
        print(f"[{func.__name__}] 函数执行耗时: {end_time - start_time:.4f} 秒")
    return wrapper

# 给下面这个有点耗时的函数套上计时装饰器
@timer_decorator
def heavy_calculation():
    print("开始进行疯狂计算...")
    total = sum(i * i for i in range(5000000))
    print("计算完成！")

# 调用函数
heavy_calculation()

x = 10  # 这是全局变量（公共公告板上写着 x=10）

def modify_global():
    # 如果没有下面这行 global x，直接写 x = 20 的话，
    # Python 只会在局部新建一个 x=20，外面的 x 依然是 10。
    
    global x  # 关键！这是“特许通行证”，告诉 Python：“我要改的是外面那个全局的 x，不要给我新建局部的！”
    x = 20    # 此时修改，真正改动了全局变量

modify_global()
print(x)  # 输出: 20

30、面向对象的编程

类，对象，继承，多态，感觉这里很简单，但是之前怎么都不理解，这就是开窍的感觉吗哈哈哈哈哈

继承和多态是为了减少代码重复和增加代码灵活性而产生

继承 (Inheritance)： 就像儿子继承父亲的财产。如果你已经写好了一个“动物”类（会吃、会睡），现在要写一个“狗”类，你不需要重新写吃和睡的方法，直接让“狗”继承“动物”，它就自动拥有了这些技能。

多态 (Polymorphism)： 意思是“同一种指令，不同的表现形态”。老板喊一句“开始工作！”，程序员开始敲代码，设计师开始画图，销售开始打电话。发出的指令一样，但不同对象执行的具体行为不同。

# 父类
class Animal:
    def speak(self):
        pass # 等待子类去具体实现

# 子类 1：狗（继承了 Animal）
class Dog(Animal):
    def speak(self):
        return "汪汪汪！"

# 子类 2：猫（继承了 Animal）
class Cat(Animal):
    def speak(self):
        return "喵喵喵！"

# --- 多态的体现 ---
def make_animal_speak(animal_obj):
    # 我不在乎传进来的是狗还是猫，我只管调用 speak()
    # 不同的动物会给出不同的反应，这就是多态！
    print(animal_obj.speak())

my_dog = Dog()
my_cat = Cat()

make_animal_speak(my_dog)  # 输出: 汪汪汪！
make_animal_speak(my_cat)  # 输出: 喵喵喵！

31、魔术方法

在 Python 里，那些名字前后都带着双下划线的方法（比如 __xxx__），被称为魔术方法（也叫 Dunder Methods）。 为什么叫魔术？ 因为它们通常不需要你手动去调用，Python 会在特定的时机“像变魔术一样”自动帮你调用。

__ init__ (初始化方法):

触发时机：*你根据“类”创建“对象”的那一瞬间，自动触发。

作用：用来给刚出生的对象绑定初始属性（比如赋予它名字、颜色）。

__ str __ (面向用户的字符串):

触发时机： 当你用 print(对象) 或者 str(对象) 时自动触发。

作用：返回一个人能看懂的、漂亮的字符串描述。如果不写这个方法，你 print(对象) 只会看到一串冰冷的内存地址（如 <__main__.Car object at 0x...>）。

__ repr__ (面向开发者的官方字符串):

触发时机：在终端里直接输入对象名回车，或者调用 repr(对象) 时触发。

作用：给程序员看的，要求极其严谨。它返回的字符串，最好能直接复制粘贴成 Python 代码来重新创建这个对象。

class Book:
    def __init__(self, title, author):
        self.title = title
        self.author = author

    # 写给普通用户看的
    def __str__(self):
        return f"《{self.title}》 作者：{self.author}"

    # 写给程序员/调试用的
    def __repr__(self):
        return f"Book(title='{self.title}', author='{self.author}')"

# 测试
my_book = Book("三体", "刘慈欣")

# 1. 触发 __str__
print(my_book)  
# 输出: 《三体》 作者：刘慈欣

# 2. 触发 __repr__ (模拟调试或直接查看对象)
print(repr(my_book)) 
# 输出: Book(title='三体', author='刘慈欣')

33、错误和异常处理

结构化捕获：try-except-finally

try放置可能引发错误的代码，此块一旦发生异常，后续代码立即终止执行，并生成一个异常对象。

except 异常捕获与处理，finally一定会被执行。

主动抛出与自定义异常：raise与继承

raise主动触发。

assert是一种调试辅助工具，语法糖等同于

if not condition: raise AssertionError(“Error message”)

声明某个条件在程序运行到此处时必须为真。如果为假，说明代码存在不可调和的逻辑漏洞。

与 try-except 的本质区别：

  • try-except 用于处理预期内可能发生、且与外部环境相关的错误（如用户输入了非法字符、网络突然断开）。这类问题程序必须能处理。
  • assert 用于捕捉预期外绝对不该发生的内部逻辑错误（如一个经过严格过滤的数组，到达此步骤时长度决不能小于0）。它是写给开发者自己看的，用于尽早暴露程序的内部 Bug。在 Python 以优化模式（-O 参数）运行时，所有的 assert 语句会被直接忽略，不影响性能。

# 🍔 【正规餐厅】普通函数 (def)
def add_normal(x, y):
    return x + y

# 调用正规餐厅
print(add_normal(3, 5))  # 输出: 8


# 🌭 【路边摊】匿名函数 (lambda)
# 因为它没名字，为了能调用它，我们临时给它贴个标签叫 add_lambda
add_lambda = lambda x, y: x + y

# 调用路边摊
print(add_lambda(3, 5))  # 输出: 8

可以作为一次性工具传给其他函数
