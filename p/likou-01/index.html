class Solution:

        '''需要在depth这个内部函数里面修改外层函数的变量ans，如果没有这句话，python会认为ans是depth函数内部的新变量，修改时会报错。'''

            return 0
        left_depth = depth(node.left)
        right_depth = depth(node.right)
        ans = max(ans,left_depth+right_depth)
        '''返回当前节点的最大深度'''
        return max(left_depth,right_depth)+1
    depth(root)
    return ans

from collections import deque
from typing import Optional, List

# Definition for a binary tree node.
# class TreeNode:
#     def __init__(self, val=0, left=None, right=None):
#         self.val = val
#         self.left = left
#         self.right = right

class Solution:
    def levelOrder(self, root: Optional[TreeNode]) -> List[List[int]]:
        # 修复：空节点直接返回空列表，符合返回值类型要求
        if not root:
            return []
        
        ans = []
        # 双向链表，用于高效地在两端追加和弹出元素 (O(1) 时间复杂度)
        queue = deque([root])
        
        # 隐式布尔值：当队列非空时继续循环
        while queue:
            level_size = len(queue)
            current_level = []
            
            # 无名氏变量 _：因为我们只需要循环指定的次数，不需要用到索引值
            for _ in range(level_size):
                node = queue.popleft()
                current_level.append(node.val)
                
                if node.left:
                    queue.append(node.left)
                if node.right:
                    queue.append(node.right)
                    
            ans.append(current_level)
            
        return ans

from typing import List, Optional

class Solution:
    def sortedArrayToBST(self, nums: List[int]) -> Optional[TreeNode]:
        def helper(left: int, right: int) -> Optional[TreeNode]:
            if left > right:
                return None

            mid = (left + right) // 2
            root = TreeNode(nums[mid])

            root.left = helper(left, mid - 1)
            root.right = helper(mid + 1, right)

            return root

        return helper(0, len(nums) - 1)

时间复杂度 O(N),空间复杂度O（logN）,二分法，递归

class Solution:
    def isValidBST(self, root: Optional[TreeNode]) -> bool:
        self.prev = None

        def inorder(node: Optional[TreeNode]) -> bool:
            if node is None:
                return True

            if not inorder(node.left):
                return False

            if self.prev is not None and node.val <= self.prev:
                return False

            self.prev = node.val

            return inorder(node.right)

        return inorder(root)

        curr = root
        while curr:
            if curr.left:
                predecessor = curr.left
                while predecessor.right:
                    predecessor  = predecessor.right
                predecessor.right = curr.right

            curr = curr.right

class Solution:
    def buildTree(self,preorder:List[int],inorder:List[int])->Optional[TreeNode]:
        inorder_map = {val:idx for idx,val in enumerate(inorder)}
        def build(pre_left: int, pre_right: int, in_left: int, in_right: int) -> Optional[TreeNode]:
            if pre_left > pre_right:
                return None
            root_val = preorder[pre_left]
            root = TreeNode(root_val)
            pindex = inorder_map[root_val]
            left_size = pindex - in_left
            root.left  = build(pre_left + 1,pre_left + left_size,in_left,pindex - 1)
            root.right = build(pre_left + left_size +1,pre_right,pindex + 1,in_right)
            return root
        n = len(preorder)
        return build(0,n-1,0,n-1)

13、路径总和

前缀和+深度优先搜索

前缀和指的是从根节点到当前节点路径上所有节点的节点和之值。

在dfs遍历的过程中，利用哈希表记录当前路径上出现过的所有前缀和及其出现的次数。

回溯：在退出当前节点的递归时，必须将当前节点的前缀和从哈希表中移除，以确保不同分支的路径互不干扰。

# Definition for a binary tree node.
# class TreeNode:
#     def __init__(self, val=0, left=None, right=None):
#         self.val = val
#         self.left = left
#         self.right = right
class Solution:
    def pathSum(self, root: Optional[TreeNode], targetSum: int) -> int:
        from collections import defaultdict
        
        # prefix_map 记录前缀和出现的次数
        prefix_map = defaultdict(int)
        # 初始化：前缀和为 0 的路径有 1 条（代表从根节点直接开始的路径）
        prefix_map[0] = 1
        
        def dfs(node, curr_sum):
            if not node:
                return 0
                
            curr_sum += node.val
            
            # 核心逻辑：查找哈希表中是否存在 curr_sum - targetSum
            # 如果存在，说明找到了符合条件的路径，累加其出现的次数
            count = prefix_map[curr_sum - targetSum]
            
            # 将当前前缀和加入哈希表，向下继续搜索
            prefix_map[curr_sum] += 1
            
            # 递归左右子树
            count += dfs(node.left, curr_sum)
            count += dfs(node.right, curr_sum)
            
            # 回溯状态：离开当前节点，消除其对其他平行分支的影响
            prefix_map[curr_sum] -= 1
            
            return count
            
        return dfs(root, 0)

        if not root or root == p or root == q:
            return root
        left = self.lowestCommonAncestor(root.left,p,q)
        right = self.lowestCommonAncestor(root.right,p,q)
        if left and right:
            return root
        return left if left else right

class Solution:
    def maxPathSum(self, root: Optional[TreeNode]) -> int:
        max_sum = float('-inf')
        def max_gain(node):
            nonlocal max_sum #声明max_sum为非局部变量，使得内部函数max_gain在执行时可以修改外部主函数中定义的max_sum
            if not node:
                return 0
            left_gain = max(max_gain(node.left),0)
            right_gain  = max(max_gain(node.right),0)
            current_path_sum = node.val + left_gain + right_gain
            max_sum = max(max_sum,current_path_sum)
            return node.val + max(left_gain,right_gain)
        max_gain(root)
        return max_sum

class Solution:

        if not grid:
            return 0
        count = 0
        rows,cols = len(grid),len(grid[0])
        
	def dfs(r: int,c:int):

            return
        grid[r][c] = '0'
        dfs(r-1,c)
        dfs(r+1,c)
        dfs(r,c-1)
        dfs(r,c+1)
    for r in range(rows):
        for c in range(cols):
            if grid[r][c] == '1':
                count += 1
                dfs(r,c)
    return count

from collections import deque
from typing import List
class Solution:
    def orangesRotting(self, grid: List[List[int]]) -> int:
        rows = len(grid)
        cols = len(grid[0])
        queue = deque()
        fresh_count = 0
        for i in range (rows):
            for c in range(cols):
                if grid[r][c] == 2:
                    queue.append((r,c))
                elif grid[r][c] == 1:
                    fresh_count += 1
        if fresh_count == 0:
            return 0
        minutes = 0
        direction = [(-1,0),(1,0),(0,-1),(0,1)]
        while queue and fresh_count > 0:
            minutes += 1
            level_size = len(queue)
            for _ in range(level_size):
                r,c = queue.popleft()
                for dr,dc in direction:
                    nr,nc = r+dr,c+dc
                    if 0<=nr <rows and 0<=nc <cols and grid[nr][nc] == 1:
                        grid[nr][nc] = 2
                        fresh_count -= 1
                        queue.append((nr,nc))
       return minutes if fresh_count == 0 else -1

from collections import deque
from typing import List
class Solution:
    def canFinish(self, numCourses: int, prerequisites: List[List[int]]) -> bool:
        adj = [[] for _ in range(numCourses)]
        in_degree = [0] * numCourses
        for course,pre in prerequisites:
            adj[pre].append(course)
            in_degree[course] += 1
        queue = deque([i for i in range(numCourses) if in_degree[i] == 0])
        completed_count = 0
        while queue:
            current = queue.popleft()
            completed_count += 1
            for next_course in adj[current]:
                in_degree[next_course] -= 1
                if in_degree[next_course] == 0:
                    queue.append(next_course)
        return completed_count == numCourses

class TrieNode:
    def __init__(self):
        # 存储子节点，使用哈希表实现字符到节点的映射
        self.children = {}
        # 标记是否为单词结尾
        self.is_end_of_word = False

class Trie:
    def __init__(self):
        # 初始化根节点，根节点本身不包含具体字符数据
        self.root = TrieNode()

    def insert(self, word: str) -> None:
        node = self.root
        for char in word:
            if char not in node.children:
                node.children[char] = TrieNode()
            node = node.children[char]
        # 单词插入完成，标记末尾节点
        node.is_end_of_word = True

    def search(self, word: str) -> bool:
        node = self.root
        for char in word:
            if char not in node.children:
                return False
            node = node.children[char]
        # 必须是单词结尾才算找到
        return node.is_end_of_word

    def startsWith(self, prefix: str) -> bool:
        node = self.root
        for char in prefix:
            if char not in node.children:
                return False
            node = node.children[char]
        # 前缀匹配成功即可
        return True

1、全排列

回溯算法，将寻找所有排列组合的过程转化为在一颗决策树上的深度优先遍历。

假设有N个空位，我们需要把N个不同的数填进去：

在第一个位置，我们有 $N$ 种选择。

填好第一个位置后，第二个位置剩下 $N-1$ 种选择。

依此类推，直到填满所有位置，就得到了一种完整的排列。

得到一种排列后，我们需要“撤销”最后一步的选择（即回退到上一个状态），去尝试其他的可能性，这个撤销的动作就是“回溯”。

class Solution:
    def permute(self, nums: List[int]) -> List[List[int]]:
        res = []
        path = []
        # 使用布尔数组记录元素的访问状态，查询时间复杂度 O(1)
        used = [False] * len(nums)
        
        def backtrack():
            # 终止条件：path 长度达到 nums 长度
            if len(path) == len(nums):
                # 必须加入 path 的拷贝，否则后续回溯会清空被引用的原对象
                res.append(path[:])
                return
            
            for i in range(len(nums)):
                # 剪枝：如果当前元素已在路径中，跳过
                if used[i]:
                    continue
                
                # 做选择
                path.append(nums[i])
                used[i] = True
                
                # 进入下一层决策树
                backtrack()
                
                # 撤销选择（回溯）
                path.pop()
                used[i] = False
                
        backtrack()
        return res

2、找子集

执行机制：

  1. 状态无条件收集：因为子集包含所有可能长度的组合，决策树上的每一个节点（无论深浅，包括根节点的空集）对应的 path 都是一个合法的子集。因此，进入递归函数的第一步，就是将当前 path 的拷贝加入结果集 res。
  2. 选择与推进：
    • 维护一个指针 start，记录当前层允许选择的起始索引。
    • 遍历从 start 到数组末尾的元素。
    • 将选中的元素加入 path，并在向下递归时，将下一层的起始位置设为 i + 1。这保证了向深层遍历时严格向后推移，切断了回头重新选择的路径，从而避免生成重复组合。
  3. 撤销选择（回溯）：递归返回后，将刚刚加入 path 的元素弹出，恢复到上一步的状态，以便在当前层继续选择下一个可用元素。

from typing import List
class Solution:
    def subsets(self, nums: List[int]) -> List[List[int]]:
        res = []
        path = []
        def backtrack(start):
            # 决策树上的每个节点都是有效子集，直接收集
            res.append(path[:])
            # 从 start 开始向后遍历，控制遍历方向
            for i in range(start, len(nums)):
                # 做选择
                path.append(nums[i])
                # 向下层推进，下一层只能从 i 的下一个位置开始选
                backtrack(i + 1)
                # 撤销选择（回溯）
                path.pop()
        backtrack(0)
        return res
